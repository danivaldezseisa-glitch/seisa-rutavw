import streamlit as st import requests import json import pandas as pd from geopy.geocoders import Nominatim import re import time # --- CONFIGURACIÓN --- # La API Key se lee desde los secretos de Streamlit. # Crea el archivo .streamlit/secrets.toml con: # OCR_API_KEY = "tu_key_aquí" try: API_KEY = st.secrets["OCR_API_KEY"] except Exception: st.error("⚠️ No se encontró la API Key. Agrega OCR_API_KEY en .streamlit/secrets.toml") st.stop() st.set_page_config(page_title="Logística OCR Masiva", layout="wide") def ocr_api_process(image_file) -> str | None: """Envía la imagen a la API de OCR.space y retorna el texto extraído.""" url = "https://api.ocr.space/parse/image" payload = { "apikey": API_KEY, "language": "spa", "isOverlayRequired": False, } # Reset del buffer antes de leer, por si ya fue leído (ej. por st.image) image_file.seek(0) try: response = requests.post(url, data=payload, files={"file": image_file}, timeout=30) response.raise_for_status() result = response.json() if result.get("OCRExitCode") == 1: parsed = result.get("ParsedResults") if parsed: return parsed[0].get("ParsedText", "").strip() else: error_msg = result.get("ErrorMessage", "Error desconocido de OCR.") st.warning(f"OCR no pudo procesar la imagen: {error_msg}") except requests.exceptions.Timeout: st.error("Error: La solicitud a la API de OCR tardó demasiado.") except requests.exceptions.RequestException as e: st.error(f"Error de conexión con la API de OCR: {e}") except (ValueError, KeyError) as e: st.error(f"Error al procesar la respuesta de la API: {e}") return None def extraer_direccion(texto: str) -> str: """ Intenta extraer una dirección del texto OCR usando patrones comunes. Busca líneas que contengan palabras clave de direcciones. """ keywords = ["calle", "av.", "avenida", "col.", "colonia", "manzana", "lote", "int.", "ext.", "piso", "depto", "cp", "c.p.", "blvd", "boulevard", "carretera", "km", "municipio"] lineas = texto.splitlines() candidatos = [] for linea in lineas: linea_lower = linea.lower() if any(kw in linea_lower for kw in keywords): candidatos.append(linea.strip()) if candidatos: return ", ".join(candidatos[:3]) # Máximo 3 líneas relevantes # Fallback: buscar líneas con números que parezcan domicilios patron_domicilio = re.compile(r'\b\d{1,5}\b') for linea in lineas: if patron_domicilio.search(linea) and len(linea.strip()) > 10: return linea.strip() return "" def obtener_coordenadas(direccion: str) -> tuple[float | None, float | None, str | None]: """Busca las coordenadas de una dirección usando Nominatim.""" if not direccion: return None, None, None geolocator = Nominatim(user_agent="dan_logistics_bot_v2") try: # Respetamos el límite de 1 req/seg de la política de Nominatim time.sleep(1) location = geolocator.geocode(direccion, timeout=10) if location: return location.latitude, location.longitude, location.address except Exception as e: st.warning(f"No se pudo geocodificar la dirección: {e}") return None, None, None # --- INTERFAZ --- st.title("📦 Sistema de Extracción Logística") st.markdown( "Sube múltiples fotos de etiquetas. " "El sistema extraerá los datos y generará el mapa automáticamente." ) uploaded_files = st.file_uploader( "Arrastra tus etiquetas aquí", type=["jpg", "png", "jpeg"], accept_multiple_files=True, ) if uploaded_files: registros = [] for uploaded_file in uploaded_files: with st.expander(f"📄 Procesando: {uploaded_file.name}", expanded=True): col1, col2 = st.columns([1, 2]) with col1: # Leemos para mostrar la imagen y luego reseteamos el buffer uploaded_file.seek(0) st.image(uploaded_file, use_column_width=True) with col2: with st.spinner("Extrayendo texto..."): # seek(0) dentro de ocr_api_process garantiza lectura fresca texto = ocr_api_process(uploaded_file) if texto: direccion_extraida = extraer_direccion(texto) lat, lon, full_addr = obtener_coordenadas(direccion_extraida) datos = { "archivo": uploaded_file.name, "texto_crudo": texto, "direccion_extraida": direccion_extraida, "lat": lat, "lon": lon, "direccion_mapa": full_addr, } registros.append(datos) st.success("¡Datos extraídos!") st.text_area("Contenido:", texto, height=100) if direccion_extraida: st.info(f"🔍 Dirección detectada: {direccion_extraida}") if lat and lon: st.info(f"📍 Ubicación confirmada: {full_addr}") st.map(pd.DataFrame({"lat": [lat], "lon": [lon]})) elif direccion_extraida: st.warning("No se encontraron coordenadas para esa dirección.") else: st.warning("No se detectó ninguna dirección en el texto.") else: st.error("No se pudo leer esta imagen.") # --- CIERRE Y EXPORTACIÓN --- if registros: st.divider() st.subheader("📊 Resumen de Procesamiento") df = pd.DataFrame(registros) st.dataframe(df) # Exportar mapa general con todos los puntos geocodificados puntos_mapa = df.dropna(subset=["lat", "lon"]) if not puntos_mapa.empty: st.subheader("🗺️ Mapa General de Envíos") st.map(puntos_mapa[["lat", "lon"]]) # Descarga del JSON completo json_string = json.dumps(registros, indent=4, ensure_ascii=False) st.download_button( label="⬇️ Descargar Base de Datos (JSON)", file_name="envios_procesados.json", mime="application/json", data=json_string, )
