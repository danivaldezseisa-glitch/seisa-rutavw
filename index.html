import streamlit as st import requests import json import pandas as pd import re from geopy.geocoders import Nominatim # --- CONFIGURACIÓN --- API_KEY = 'K82391950088957' st.set_page_config(page_title="Logística OCR Masiva", layout="wide") def ocr_api_process(image_file): """Envía la imagen a la API de OCR.space""" url = "https://api.ocr.space/parse/image" payload = {'apikey': API_KEY, 'language': 'spa', 'isOverlayRequired': False} files = {'file': image_file} try: response = requests.post(url, data=payload, files=files) result = response.json() if result.get("OCRExitCode") == 1: return result.get("ParsedResults")[0].get("ParsedText") except Exception as e: st.error(f"Error de conexión: {e}") return None def extraer_tracking(texto): """Intenta extraer el número de tracking del texto OCR""" if not texto: return None # Patrones comunes de tracking: secuencias alfanuméricas de 10-30 caracteres patrones = [ r'\b([A-Z]{2}\d{9}[A-Z]{2})\b', # Formato postal internacional r'\b(\d{12,22})\b', # Numérico largo (FedEx, UPS, DHL) r'\b([A-Z0-9]{10,30})\b', # Alfanumérico genérico ] for patron in patrones: match = re.search(patron, texto) if match: return match.group(1) # Fallback: devolver la primera línea no vacía como tracking for linea in texto.splitlines(): linea = linea.strip() if len(linea) >= 8: return linea return texto.strip() def obtener_coordenadas(texto): """Busca la dirección en el mapa""" geolocator = Nominatim(user_agent="dan_logistics_bot_v2") try: location = geolocator.geocode(texto, timeout=10) if location: return location.latitude, location.longitude, location.address except: pass return None, None, None def buscar_en_archivo(tracking, df_archivo): """Busca el tracking en el DataFrame del archivo cargado""" if df_archivo is None or tracking is None: return None tracking_limpio = str(tracking).strip().upper() for col in df_archivo.columns: col_upper = df_archivo[col].astype(str).str.strip().str.upper() fila = df_archivo[col_upper == tracking_limpio] if not fila.empty: return fila.iloc[0] return None def cargar_archivo_datos(archivo): """Carga el archivo de datos (CSV, Excel o JSON)""" nombre = archivo.name.lower() try: if nombre.endswith('.csv'): return pd.read_csv(archivo) elif nombre.endswith(('.xlsx', '.xls')): return pd.read_excel(archivo) elif nombre.endswith('.json'): return pd.read_json(archivo) except Exception as e: st.error(f"Error al leer el archivo: {e}") return None # ─── COLUMNAS DE DATOS ESPERADAS (ajusta según tu archivo) ──────────────────── COL_CLIENTE = ["nombre", "cliente", "name", "customer"] COL_TELEFONO = ["telefono", "teléfono", "phone", "tel", "celular", "movil", "móvil"] COL_PEDIDO = ["pedido", "order", "orden", "numero_pedido", "order_id", "id"] COL_DIRECCION = ["direccion", "dirección", "address", "domicilio"] def detectar_columna(df, candidatos): for c in candidatos: for col in df.columns: if c in col.lower(): return col return None # ─── INTERFAZ ──────────────────────────────────────────────────────────────── st.title("📦 Sistema de Extracción Logística") st.markdown("Sube múltiples fotos de etiquetas. El sistema extraerá el tracking y cruzará los datos automáticamente.") # ── PASO 1: Escaneo de guías ────────────────────────────────────────────────── st.header("📷 Paso 1 — Escanear Guías") uploaded_files = st.file_uploader( "Arrastra tus etiquetas aquí", type=['jpg', 'png', 'jpeg'], accept_multiple_files=True, key="guias" ) # Guardamos trackings escaneados en session_state if "trackings_escaneados" not in st.session_state: st.session_state.trackings_escaneados = {} # {nombre_archivo: tracking} if "fotos_articulos" not in st.session_state: st.session_state.fotos_articulos = {} # {tracking: [foto1, foto2, ...]} if uploaded_files: for uploaded_file in uploaded_files: nombre = uploaded_file.name if nombre not in st.session_state.trackings_escaneados: with st.expander(f"📄 {nombre}", expanded=True): col1, col2 = st.columns([1, 2]) with col1: st.image(uploaded_file, use_container_width=True) with col2: with st.spinner("Extrayendo tracking..."): texto = ocr_api_process(uploaded_file) tracking = extraer_tracking(texto) if tracking: st.session_state.trackings_escaneados[nombre] = tracking st.success(f"✅ Tracking detectado:") st.code(tracking) else: st.error("No se pudo extraer tracking de esta imagen.") else: with st.expander(f"📄 {nombre} ✅", expanded=False): st.code(st.session_state.trackings_escaneados[nombre]) # ── PASO 2: Subir archivo de datos ──────────────────────────────────────────── st.divider() st.header("📁 Paso 2 — Subir Archivo de Datos") st.caption("El archivo debe contener columnas de tracking, nombre del cliente, teléfono, número de pedido y dirección.") archivo_datos = st.file_uploader( "Sube tu archivo (CSV, Excel o JSON)", type=['csv', 'xlsx', 'xls', 'json'], key="archivo_datos" ) df_datos = None if archivo_datos: df_datos = cargar_archivo_datos(archivo_datos) if df_datos is not None: st.success(f"✅ Archivo cargado: {len(df_datos)} registros, {len(df_datos.columns)} columnas.") with st.expander("Vista previa del archivo"): st.dataframe(df_datos.head(5)) # ── PASO 3: Cargar datos ────────────────────────────────────────────────────── st.divider() st.header("🚀 Paso 3 — Cargar Datos") btn_cargar = st.button("Cargar y Cruzar Datos", type="primary", disabled=(df_datos is None or len(st.session_state.trackings_escaneados) == 0)) if btn_cargar and df_datos is not None: col_cliente = detectar_columna(df_datos, COL_CLIENTE) col_telefono = detectar_columna(df_datos, COL_TELEFONO) col_pedido = detectar_columna(df_datos, COL_PEDIDO) col_direccion = detectar_columna(df_datos, COL_DIRECCION) registros = [] for nombre_archivo, tracking in st.session_state.trackings_escaneados.items(): fila = buscar_en_archivo(tracking, df_datos) dato = { "archivo": nombre_archivo, "tracking": tracking, "cliente": fila[col_cliente] if (fila is not None and col_cliente) else "—", "telefono": fila[col_telefono] if (fila is not None and col_telefono) else "—", "pedido": fila[col_pedido] if (fila is not None and col_pedido) else "—", "direccion": fila[col_direccion] if (fila is not None and col_direccion) else "—", } registros.append(dato) st.session_state.registros_finales = registros st.success("✅ Datos cruzados correctamente.") # ── RESULTADOS POR CLIENTE ──────────────────────────────────────────────────── if "registros_finales" in st.session_state and st.session_state.registros_finales: st.divider() st.subheader("👤 Detalle por Cliente") for reg in st.session_state.registros_finales: tracking = reg["tracking"] with st.expander(f"📦 Tracking: {tracking} | {reg['cliente']}", expanded=True): # Datos del cliente col_a, col_b = st.columns(2) with col_a: st.markdown(f"**🙍 Cliente:** {reg['cliente']}") st.markdown(f"**📞 Teléfono:** {reg['telefono']}") with col_b: st.markdown(f"**🛒 Pedido:** {reg['pedido']}") st.markdown(f"**📍 Dirección:** {reg['direccion']}") # Mapa si hay dirección válida if reg["direccion"] not in ("—", None, ""): with st.spinner("Buscando ubicación..."): lat, lon, full_addr = obtener_coordenadas(reg["direccion"]) if lat: st.info(f"🗺️ {full_addr}") st.map(pd.DataFrame({'lat': [lat], 'lon': [lon]})) # ── FOTOS DEL ARTÍCULO ────────────────────────────────────────── st.markdown("---") st.markdown("#### 📸 Fotos del Artículo") key_uploader = f"fotos_{tracking}" fotos_nuevas = st.file_uploader( "Sube fotos del artículo para este tracking", type=['jpg', 'png', 'jpeg'], accept_multiple_files=True, key=key_uploader ) if fotos_nuevas: st.session_state.fotos_articulos[tracking] = fotos_nuevas fotos_guardadas = st.session_state.fotos_articulos.get(tracking, []) if fotos_guardadas: cols_fotos = st.columns(min(len(fotos_guardadas), 4)) for i, foto in enumerate(fotos_guardadas): with cols_fotos[i % 4]: st.image(foto, use_container_width=True, caption=foto.name) # ── EXPORTACIÓN ────────────────────────────────────────────────────────── st.divider() st.subheader("📊 Resumen de Procesamiento") df_resumen = pd.DataFrame(st.session_state.registros_finales) st.dataframe(df_resumen) json_string = json.dumps(st.session_state.registros_finales, indent=4, ensure_ascii=False) st.download_button( label="⬇️ Descargar Base de Datos (JSON)", file_name="envios_procesados.json", mime="application/json", data=json_string )
